def NeggersStanley.posetEulerianPolynomial : {n : ℕ} → PartialOrder (Fin n) → Polynomial ℝ :=
fun {n} P =>
  (List.map (fun l => Polynomial.X ^ NeggersStanley.descents l)
      (List.filter (fun l => decide (List.Pairwise (fun a b => ¬b < a) l)) (List.finRange n).permutations)).sum
def NeggersStanley.NeggersConjecture : Prop :=
∀ (n : ℕ) (P : NeggersStanley.LabeledPoset n),
  P.NaturallyLabeled →
    ∀ (z : ℂ), (Polynomial.map (algebraMap ℝ ℂ) (NeggersStanley.descentPolynomial P)).IsRoot z → z.im = 0
def NeggersStanley.descentPolynomial : {n : ℕ} → NeggersStanley.LabeledPoset n → Polynomial ℝ :=
fun {n} P =>
  (List.map (fun l => Polynomial.X ^ (1 + NeggersStanley.descents l)) (NeggersStanley.linearExtensionWords P)).sum
def NeggersStanley.linearExtensionWords : {n : ℕ} → NeggersStanley.LabeledPoset n → List (List (Fin n)) :=
fun {n} P => List.filter (fun l => decide (List.Pairwise (fun a b => ¬P.rel b a) l)) (List.finRange n).permutations
